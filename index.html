import pygame import random import sys pygame.init() # Ekran ayarları WIDTH, HEIGHT = 1000, 700 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("👑 KRAL MEYVƏ SLOT - 1Win Stil 👑") # Rənglər BLACK = (0, 0, 0) GOLD = (255, 215, 0) RED = (255, 50, 50) GREEN = (50, 255, 50) DARK_BLUE = (10, 20, 60) WHITE = (255, 255, 255) # Fontlar big_font = pygame.font.SysFont("Arial", 60, bold=True) font = pygame.font.SysFont("Arial", 36) small_font = pygame.font.SysFont("Arial", 24) # Simvollar və ödənişlər SYMBOLS = ['🍒', '🍋', '🍊', '🍇', '🛎️', 'BAR', '7️⃣', '⭐'] PAYOUTS = {'🍒': 20, '🍋': 30, '🍊': 40, '🍇': 50, '🛎️': 80, 'BAR': 150, '7️⃣': 300, '⭐': 500} class Button: def __init__(self, x, y, w, h, text, color): self.rect = pygame.Rect(x, y, w, h) self.text = text self.color = color def draw(self): pygame.draw.rect(screen, self.color, self.rect, border_radius=12) pygame.draw.rect(screen, GOLD, self.rect, 5, border_radius=12) text_surf = font.render(self.text, True, BLACK) text_rect = text_surf.get_rect(center=self.rect.center) screen.blit(text_surf, text_rect) def is_clicked(self, pos): return self.rect.collidepoint(pos) # Oyun dəyişənləri balance = 5000 bet = 100 reels = [[random.choice(SYMBOLS) for _ in range(3)] for _ in range(3)] spin_button = Button(410, 550, 180, 80, "SPIN!", GREEN) bet_up = Button(680, 555, 65, 65, "+", GOLD) bet_down = Button(760, 555, 65, 65, "-", GOLD) def draw_reels(): for i in range(3): for j in range(3): x = 170 + i * 210 y = 130 + j * 115 pygame.draw.rect(screen, (25, 25, 70), (x-15, y-15, 185, 115), border_radius=15) sym_text = big_font.render(reels[i][j], True, GOLD if reels[i][j] in ['7️⃣', 'BAR', '⭐'] else WHITE) screen.blit(sym_text, (x + 35, y + 10)) def calculate_win(): win = 0 for row in range(3): # 3 xətt (üst, orta, alt) if reels[0][row] == reels[1][row] == reels[2][row]: symbol = reels[0][row] multiplier = PAYOUTS.get(symbol, 10) win += multiplier * (bet // 10) return win def spin_animation(): global reels # Sürətli spin for _ in range(14): for i in range(3): reels[i] = [random.choice(SYMBOLS) for _ in range(3)] draw_game(0) pygame.display.flip() pygame.time.delay(55) # Reel-reel dayanma for r in range(3): target = [random.choice(SYMBOLS) for _ in range(3)] for _ in range(8): reels[r] = [random.choice(SYMBOLS) for _ in range(3)] draw_game(0) pygame.display.flip() pygame.time.delay(95 + r*25) reels[r] = target draw_game(0) pygame.display.flip() pygame.time.delay(350) def draw_game(current_win=0): screen.fill(DARK_BLUE) # Başlıq title = big_font.render("👑 KRAL SLOT 👑", True, GOLD) screen.blit(title, (WIDTH//2 - title.get_width()//2, 25)) # Balans bal_text = font.render(f"Balans: {balance} AZN", True, GOLD) screen.blit(bal_text, (70, 85)) bet_text = font.render(f"Mərc: {bet}", True, GOLD) screen.blit(bet_text, (WIDTH - 320, 85)) draw_reels() if current_win > 0: win_text = big_font.render(f"🎉 QAZANDIN: {current_win}! 🎉", True, GREEN) screen.blit(win_text, (WIDTH//2 - win_text.get_width()//2, 475)) spin_button.draw() bet_up.draw() bet_down.draw() # Kiçik məlumat info = small_font.render("3 eyni simvol = QAZANMA! 👑", True, (200,200,200)) screen.blit(info, (WIDTH//2 - info.get_width()//2, 640)) pygame.display.flip() # Əsas oyun dövrəsi clock = pygame.time.Clock() running = True current_win = 0 while running: for event in pygame.event.get(): if event.type == pygame.QUIT: running = False if event.type == pygame.MOUSEBUTTONDOWN: pos = pygame.mouse.get_pos() if spin_button.is_clicked(pos) and balance >= bet: balance -= bet current_win = 0 spin_animation() current_win = calculate_win() if current_win > 0: balance += current_win elif bet_up.is_clicked(pos) and bet < 2000: bet += 50 elif bet_down.is_clicked(pos) and bet > 50: bet -= 50 draw_game(current_win) clock.tick(60) pygame.quit() sys.exit()
