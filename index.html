import pygame import random import sys pygame.init() # Ekran WIDTH, HEIGHT = 1000, 700 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("👑 MÜKƏMMƏL KRAL SLOT - 1Win Edition 👑") # Rənglər BLACK = (0, 0, 0) GOLD = (255, 215, 0) RED = (255, 50, 50) GREEN = (50, 255, 50) GRAY = (100, 100, 100) DARK_BLUE = (10, 20, 60) WHITE = (255, 255, 255) # Fontlar big_font = pygame.font.SysFont("Arial", 65, bold=True) font = pygame.font.SysFont("Arial", 38) small_font = pygame.font.SysFont("Arial", 26) # Simvollar (hamısı emoji - heç bir səhv çıxmayacaq) SYMBOLS = ['🍒', '🍋', '🍊', '🍇', '🛎️', '💎', '7️⃣', '⭐'] PAYOUTS = {'🍒': 25, '🍋': 35, '🍊': 45, '🍇': 60, '🛎️': 90, '💎': 180, '7️⃣': 350, '⭐': 600} class Button: def __init__(self, x, y, w, h, text): self.rect = pygame.Rect(x, y, w, h) self.text = text def draw(self, color): pygame.draw.rect(screen, color, self.rect, border_radius=15) pygame.draw.rect(screen, GOLD, self.rect, 6, border_radius=15) text_surf = font.render(self.text, True, BLACK) screen.blit(text_surf, text_surf.get_rect(center=self.rect.center)) def is_clicked(self, pos): return self.rect.collidepoint(pos) # Oyun dəyişənləri balance = 10000 # daha çox başlanğıc bet = 100 reels = [[random.choice(SYMBOLS) for _ in range(3)] for _ in range(3)] is_spinning = False current_win = 0 win_time = 0 win_display_duration = 3000 # 3 saniyə spin_button = Button(410, 550, 180, 85, "SPIN!") bet_up = Button(680, 555, 70, 70, "+") bet_down = Button(760, 555, 70, 70, "-") def draw_reels(): for i in range(3): for j in range(3): x = 165 + i * 225 y = 130 + j * 118 # Qızıl çərçivə pygame.draw.rect(screen, (40, 30, 10), (x-20, y-20, 205, 128), border_radius=20) pygame.draw.rect(screen, GOLD, (x-20, y-20, 205, 128), 8, border_radius=20) color = GOLD if reels[i][j] in ['7️⃣', '💎', '⭐'] else WHITE sym_text = big_font.render(reels[i][j], True, color) screen.blit(sym_text, (x + 45, y + 15)) def calculate_win(): win = 0 # 3 üfüqi xətt for row in range(3): if reels[0][row] == reels[1][row] == reels[2][row]: sym = reels[0][row] win += PAYOUTS.get(sym, 10) * (bet // 10) # 2 diagonal if reels[0][0] == reels[1][1] == reels[2][2]: sym = reels[0][0] win += PAYOUTS.get(sym, 10) * (bet // 10) if reels[0][2] == reels[1][1] == reels[2][0]: sym = reels[0][2] win += PAYOUTS.get(sym, 10) * (bet // 10) return win def spin_animation(): global reels # Sürətli spin for _ in range(16): for i in range(3): reels[i] = [random.choice(SYMBOLS) for _ in range(3)] draw_game() pygame.display.flip() pygame.time.delay(45) # Reel-reel dayanma (real kazino effekti) for r in range(3): target = [random.choice(SYMBOLS) for _ in range(3)] for _ in range(9): reels[r] = [random.choice(SYMBOLS) for _ in range(3)] draw_game() pygame.display.flip() pygame.time.delay(80 + r * 30) reels[r] = target draw_game() pygame.display.flip() pygame.time.delay(320) def draw_game(): screen.fill(DARK_BLUE) # Başlıq title = big_font.render("👑 KRAL SLOT 👑", True, GOLD) screen.blit(title, (WIDTH//2 - title.get_width()//2, 25)) # Balans və mərc bal_text = font.render(f"Balans: {balance:,} AZN", True, GOLD) screen.blit(bal_text, (60, 90)) bet_text = font.render(f"Mərc: {bet}", True, GOLD) screen.blit(bet_text, (WIDTH - 340, 90)) draw_reels() # Qazanma yazısı (3 saniyə sonra yox olur) if current_win > 0 and pygame.time.get_ticks() - win_time < win_display_duration: if current_win >= 5000: # JACKPOT effekti win_surf = big_font.render(f"🎊 JACKPOT! {current_win} AZN 🎊", True, (255, 215, 0)) else: win_surf = big_font.render(f"🎉 QAZANDIN: {current_win} AZN! 🎉", True, GREEN) screen.blit(win_surf, (WIDTH//2 - win_surf.get_width()//2, 465)) else: global current_win current_win = 0 # Spin düyməsi (balans kifayət etməyəndə boz + spin zamanı blok) spin_color = GRAY if balance < bet or is_spinning else GREEN spin_button.draw(spin_color) bet_up.draw(GOLD) bet_down.draw(GOLD) info = small_font.render("3 eyni simvol = QAZAN! 👑 Diagonal da işləyir!", True, (200,200,200)) screen.blit(info, (WIDTH//2 - info.get_width()//2, 645)) pygame.display.flip() # Əsas dövrə clock = pygame.time.Clock() running = True global win_time win_time = 0 while running: for event in pygame.event.get(): if event.type == pygame.QUIT: running = False if event.type == pygame.MOUSEBUTTONDOWN and not is_spinning: pos = pygame.mouse.get_pos() if spin_button.is_clicked(pos) and balance >= bet: balance -= bet is_spinning = True current_win = 0 spin_animation() current_win = calculate_win() if current_win > 0: balance += current_win win_time = pygame.time.get_ticks() is_spinning = False elif bet_up.is_clicked(pos) and bet < 5000: bet += 50 elif bet_down.is_clicked(pos) and bet > 50: bet -= 50 draw_game() clock.tick(60) pygame.quit() sys.exit()
